#include <stdio.h>

int main() {
    int n, no, pages[30], frame[10];
    int count = 0, flag1, flag2, flag3;
    int i, j, k, pos, max, future[10];

    printf("Enter number of pages: ");
    scanf("%d", &n);

    printf("Enter the page reference string:\n");
    for (i = 0; i < n; i++)
        scanf("%d", &pages[i]);

    printf("Enter number of frames (minimum 3): ");
    scanf("%d", &no);

    for (i = 0; i < no; i++)
        frame[i] = -1;

    printf("\nPage Reference String\tFrames\t\t\tStatus\n");

    for (i = 0; i < n; i++) {
        flag1 = flag2 = 0;

        // check if page already exists
        for (j = 0; j < no; j++) {
            if (frame[j] == pages[i]) {
                flag1 = flag2 = 1;
                break;
            }
        }

        // if page not found
        if (flag1 == 0) {
            for (j = 0; j < no; j++) {
                if (frame[j] == -1) {
                    frame[j] = pages[i];
                    flag2 = 1;
                    count++;
                    break;
                }
            }
        }

        // if all frames are full
        if (flag2 == 0) {
            for (j = 0; j < no; j++) {
                future[j] = -1;
                for (k = i + 1; k < n; k++) {
                    if (frame[j] == pages[k]) {
                        future[j] = k;
                        break;
                    }
                }
            }

            pos = -1;
            max = -1;

            for (j = 0; j < no; j++) {
                if (future[j] == -1) {
                    pos = j;
                    break;
                }
                if (future[j] > max) {
                    max = future[j];
                    pos = j;
                }
            }

            frame[pos] = pages[i];
            count++;
        }

        // display frame content
        printf("%d\t\t\t", pages[i]);
        for (k = 0; k < no; k++)
            printf("%d ", frame[k]);
        if (flag1 == 1)
            printf("\tNo Fault\n");
        else
            printf("\tPage Fault\n");
    }

    printf("\nTotal Page Faults = %d\n", count);
    return 0;
}
