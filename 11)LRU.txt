#include <stdio.h>

int main() {
    int n, no, pages[30], frame[10], count = 0;
    int time[10]; // to keep track of usage time
    int i, j, k, min, pos, flag1, flag2;
    int t = 0; // time counter

    printf("Enter number of pages: ");
    scanf("%d", &n);

    printf("Enter the page reference string:\n");
    for (i = 0; i < n; i++)
        scanf("%d", &pages[i]);

    printf("Enter number of frames (minimum 3): ");
    scanf("%d", &no);

    for (i = 0; i < no; i++) {
        frame[i] = -1;
        time[i] = 0;
    }

    printf("\nPage Reference String\tFrames\t\t\tStatus\n");

    for (i = 0; i < n; i++) {
        flag1 = flag2 = 0;

        // check if page already exists
        for (j = 0; j < no; j++) {
            if (frame[j] == pages[i]) {
                flag1 = flag2 = 1;
                t++;
                time[j] = t; // update recent usage time
                break;
            }
        }

        // if page not found in any frame
        if (flag1 == 0) {
            for (j = 0; j < no; j++) {
                if (frame[j] == -1) {
                    t++;
                    count++;
                    frame[j] = pages[i];
                    time[j] = t;
                    flag2 = 1;
                    break;
                }
            }
        }

        // if all frames are full
        if (flag2 == 0) {
            min = time[0];
            pos = 0;
            for (j = 1; j < no; j++) {
                if (time[j] < min) {
                    min = time[j];
                    pos = j;
                }
            }
            t++;
            frame[pos] = pages[i];
            time[pos] = t;
            count++;
        }

        // print current frame status
        printf("%d\t\t\t", pages[i]);
        for (k = 0; k < no; k++)
            printf("%d ", frame[k]);
        if (flag1 == 1)
            printf("\tNo Fault\n");
        else
            printf("\tPage Fault\n");
    }

    printf("\nTotal Page Faults = %d\n", count);
    return 0;
}
